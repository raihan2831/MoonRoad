#include <Arduino.h>
#include <SPI.h>
#include <SDFS.h>

// --- EXTERNAL SD NAND SPI PINS ---
const int NAND_CS_PIN  = 17; 
const int NAND_SCK_PIN = 18; 
const int NAND_TX_PIN  = 19; 
const int NAND_RX_PIN  = 16; 

void setup() {
    Serial.begin(115200);
    delay(3000);

    Serial.println("\n=================================");
    Serial.println("📂 SD NAND DATA EXTRACTOR");
    Serial.println("=================================");

    // 1. Initialize SPI Bus
    SPI.setSCK(NAND_SCK_PIN);
    SPI.setTX(NAND_TX_PIN);
    SPI.setRX(NAND_RX_PIN);
    SPI.begin();

    // 2. Configure SDFS for the external SD NAND module
    SDFSConfig sdConfig;
    sdConfig.setCSPin(NAND_CS_PIN);
    SDFS.setConfig(sdConfig);

    if (!SDFS.begin()) {
        Serial.println("❌ Failed to mount SD NAND! Check wiring.");
        return;
    }
    Serial.println("✅ SD NAND Mounted Successfully.\n");

    // 3. Open the flight log directly in read mode ("r")
    File logFile = SDFS.open("/flight_log.csv", "r");
    if (!logFile) {
        Serial.println("❌ Error: /flight_log.csv not found on SD NAND.");
        Serial.println("⚠️ Make sure you have run your main telemetry logging script at least once!");
        return;
    }

    Serial.println("--- START OF CSV DATA ---");
    while (logFile.available()) {
        Serial.write(logFile.read()); // Streams raw CSV content to terminal
    }
    Serial.println("\n--- END OF CSV DATA ---");

    logFile.close();
}

void loop() {
    // Nothing to do here
}
